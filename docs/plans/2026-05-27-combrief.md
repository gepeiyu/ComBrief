# ComBrief Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use tre:subagent-driven-development (recommended) or tre:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 交付 macOS + Windows 托盘应用 ComBrief，为 Cursor 与 Claude Code 各显示一盏可动画状态灯，通过最小侵入 hook bridge 上报四态并触发红灯通知。

**Architecture:** Electron 单体（Main 进程）内嵌 `127.0.0.1` HTTP 服务、纯函数状态机、按 `appId` 管理的 `Tray` 图标；hook 逻辑外挂于 `~/.combrief/apps/<appId>/bridge.mjs`，用户配置仅增加 bridge 条目。Cursor 改 `~/.cursor/hooks.json`；Claude Code 改 `~/.claude/settings.json` 的 `hooks` 字段（官方无独立 `hooks.json`）。

**Tech Stack:** Electron 33+、TypeScript 5、Vitest、electron-builder、Node 20+

**Spec:** [docs/specs/2026-05-27-combrief-design.md](../specs/2026-05-27-combrief-design.md)

```typescript
export type LightStatus = 'idle' | 'working' | 'waiting_user' | 'offline';

export type StateEvent =
  | 'sessionStart'
  | 'sessionEnd'
  | 'beforeSubmitPrompt'
  | 'preToolUse'
  | 'postToolUse'
  | 'stop'
  | 'afterAgentResponse'
  | 'heartbeat';

export interface AppState {
  status: LightStatus;
  lastEventAt: number;
  lastHeartbeatAt: number;
}

const WORKING_EVENTS: StateEvent[] = [
  'sessionStart',
  'beforeSubmitPrompt',
  'preToolUse',
  'postToolUse',
];

export function reduceState(
  current: LightStatus,
  event: StateEvent,
): LightStatus {
  if (event === 'sessionEnd') return 'offline';
  if (event === 'stop' || event === 'afterAgentResponse') return 'waiting_user';
  if (WORKING_EVENTS.includes(event)) return 'working';
  if (event === 'heartbeat') {
    if (current === 'offline') return 'idle';
    return current;
  }
  return current;
}

export function applyHeartbeatTimeout(
  app: AppState,
  timeoutMs: number,
  now = Date.now(),
): AppState {
  if (now - app.lastHeartbeatAt >= timeoutMs) {
    return { ...app, status: 'offline' };
  }
  return app;
}
```

```typescript
import type { LightStatus } from './state-machine';

export class NotificationService {
  private lastNotifiedAt = new Map<string, number>();

  constructor(private debounceMs: number) {}

  shouldNotify(
    appId: string,
    next: LightStatus,
    prev: LightStatus,
  ): boolean {
    if (next !== 'waiting_user') return false;
    if (prev !== 'waiting_user') return true;
    const last = this.lastNotifiedAt.get(appId) ?? 0;
    return Date.now() - last >= this.debounceMs;
  }

  markNotified(appId: string): void {
    this.lastNotifiedAt.set(appId, Date.now());
  }
}
```

import { existsSync, mkdirSync, readFileSync, writeFileSync } from 'node:fs';
import { join } from 'node:path';
import { randomBytes } from 'node:crypto';

export interface CombriefConfig {
  port: number;
  token: string;
  heartbeatTimeoutMs: number;
  idleAfterWorkingMs: number;
  notificationsEnabled: boolean;
  launchAtLogin: boolean;
  apps: string[];
}

export function combriefHome(): string {
  return join(process.env.HOME ?? process.env.USERPROFILE ?? '', '.combrief');
}

export function defaultConfig(): CombriefConfig {
  return {
    port: 3847,
    token: randomBytes(24).toString('hex'),
    heartbeatTimeoutMs: 45_000,
    idleAfterWorkingMs: 5_000,
    notificationsEnabled: true,
    launchAtLogin: false,
    apps: [],
  };
}

export function loadConfig(home = combriefHome()): CombriefConfig {
  const path = join(home, 'config.json');
  if (!existsSync(path)) return defaultConfig();
  const raw = JSON.parse(readFileSync(path, 'utf8')) as Partial<CombriefConfig>;
  return { ...defaultConfig(), ...raw, token: raw.token ?? defaultConfig().token };
}

export function saveConfig(home: string, cfg: CombriefConfig): void {
  mkdirSync(home, { recursive: true });
  writeFileSync(join(home, 'config.json'), JSON.stringify(cfg, null, 2));
}

function request(
  server: http.Server,
  method: string,
  path: string,
  body?: object,
  token?: string,
): Promise<{ status: number; json: unknown }> {
  const addr = server.address();
  if (!addr || typeof addr === 'string') throw new Error('no addr');
  return new Promise((resolve, reject) => {
    const data = body ? JSON.stringify(body) : undefined;
    const req = http.request(
      {
        host: '127.0.0.1',
        port: addr.port,
        method,
        path,
        headers: {
          'Content-Type': 'application/json',
          ...(token ? { Authorization: `Bearer ${token}` } : {}),
        },
      },
      (res) => {
        let raw = '';
        res.on('data', (c) => (raw += c));
        res.on('end', () =>
          resolve({ status: res.statusCode ?? 0, json: JSON.parse(raw || '{}') }),
        );
      },
    );
    req.on('error', reject);
    if (data) req.write(data);
    req.end();
  });
}

describe('http-server', () => {

```typescript
import http from 'node:http';
import type { StateEvent } from './state-machine';

export interface ServerOptions {
  port: number;
  token: string;
  registeredApps: Set<string>;
  onState: (payload: {
    appId: string;
    event: StateEvent;
    sessionId?: string;
    timestamp: number;
  }) => void;
}

const VALID_EVENTS = new Set<string>([
  'sessionStart',
  'sessionEnd',
  'beforeSubmitPrompt',
  'preToolUse',
  'postToolUse',
  'stop',
  'afterAgentResponse',
  'heartbeat',
]);

export function createCombriefServer(
  opts: ServerOptions,
): http.Server {
  return http.createServer((req, res) => {
    const auth = req.headers.authorization;
    if (auth !== `Bearer ${opts.token}`) {
      res.writeHead(401).end(JSON.stringify({ ok: false }));
      return;
    }
    if (req.method === 'GET' && req.url === '/v1/health') {
      res.writeHead(200, { 'Content-Type': 'application/json' });
      res.end(JSON.stringify({ ok: true, version: '0.1.0' }));
      return;
    }
    if (req.method === 'POST' && req.url === '/v1/state') {
      let body = '';
      req.on('data', (c) => (body += c));
      req.on('end', () => {
        try {
          const parsed = JSON.parse(body) as {
            appId: string;
            event: string;
            sessionId?: string;
            timestamp: number;
          };
          if (!opts.registeredApps.has(parsed.appId)) {
            res.writeHead(400).end(JSON.stringify({ ok: false }));
            return;
          }
          if (!VALID_EVENTS.has(parsed.event)) {
            res.writeHead(400).end(JSON.stringify({ ok: false }));
            return;
          }
          opts.onState({
            appId: parsed.appId,
            event: parsed.event as StateEvent,
            sessionId: parsed.sessionId,
            timestamp: parsed.timestamp ?? Date.now(),
          });
          res.writeHead(200).end(JSON.stringify({ ok: true }));
        } catch {
          res.writeHead(400).end(JSON.stringify({ ok: false }));
        }
      });
      return;
    }
    res.writeHead(404).end();
  });
}
```

import type { StateEvent } from '../state-machine';

const CLAUDE_MAP: Record<string, StateEvent> = {
  SessionStart: 'sessionStart',
  SessionEnd: 'sessionEnd',
  UserPromptSubmit: 'beforeSubmitPrompt',
  PreToolUse: 'preToolUse',
  PostToolUse: 'postToolUse',
  Stop: 'stop',
};

export function normalizeHookEvent(
  appId: string,
  raw: string,
): StateEvent | null {
  if (appId === 'cursor') {
    const allowed: StateEvent[] = [
      'sessionStart',
      'sessionEnd',
      'beforeSubmitPrompt',
      'preToolUse',
      'postToolUse',
      'stop',
      'afterAgentResponse',
    ];
    return allowed.includes(raw as StateEvent) ? (raw as StateEvent) : null;
  }
  if (appId === 'claude-code') {
    return CLAUDE_MAP[raw] ?? null;
  }
  return null;
}

```javascript
#!/usr/bin/env node
import { readFileSync, appendFileSync, mkdirSync } from 'node:fs';
import { spawnSync } from 'node:child_process';
import { join } from 'node:path';
import { homedir } from 'node:os';

const APP_ID = 'cursor';
const home = join(homedir(), '.combrief');
const config = JSON.parse(
  readFileSync(join(home, 'config.json'), 'utf8'),
);
const input = JSON.parse(readFileSync(0, 'utf8'));
const hookEvent = process.env.CURSOR_HOOK_EVENT ?? input.hook_event ?? 'heartbeat';

async function report(event) {
  try {
    await fetch(`http://127.0.0.1:${config.port}/v1/state`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        Authorization: `Bearer ${config.token}`,
      },
      body: JSON.stringify({
        appId: APP_ID,
        event,
        timestamp: Date.now(),
        sessionId: input.conversation_id,
      }),
      signal: AbortSignal.timeout(500),
    });
  } catch (err) {
    mkdirSync(join(home, 'logs'), { recursive: true });
    appendFileSync(
      join(home, 'logs', 'bridge.log'),
      `${new Date().toISOString()} cursor report failed: ${err}\n`,
    );
  }
}

const EVENT_MAP = {
  sessionStart: 'sessionStart',
  sessionEnd: 'sessionEnd',
  beforeSubmitPrompt: 'beforeSubmitPrompt',
  preToolUse: 'preToolUse',
  postToolUse: 'postToolUse',
  stop: 'stop',
  afterAgentResponse: 'afterAgentResponse',
};
const normalized = EVENT_MAP[hookEvent];
if (normalized) await report(normalized);

const chainPath = join(home, 'apps', APP_ID, 'chain.json');
if (require('node:fs').existsSync(chainPath)) {
  const chain = JSON.parse(readFileSync(chainPath, 'utf8'));
  for (const cmd of chain.commands ?? []) {
    const r = spawnSync(cmd, { input: JSON.stringify(input), encoding: 'utf8', shell: true });
    if (r.stdout) process.stdout.write(r.stdout);
    if (r.stderr) process.stderr.write(r.stderr);
    process.exitCode = r.status ?? 0;
  }
}
process.exit(process.exitCode ?? 0);
```

- [ ] **Step 2: 实现（跟踪事件列表）**

```typescript
const CURSOR_EVENTS = [
  'sessionStart',
  'sessionEnd',
  'beforeSubmitPrompt',
  'preToolUse',
  'postToolUse',
  'stop',
  'afterAgentResponse',
] as const;

const MARKER = 'combrief-bridge';

export function injectCursorBridge(
  hooksJson: { version: number; hooks: Record<string, { command: string; env?: Record<string, string> }[]> },
  bridgePath: string,
  appId: string,
) {
  const next = structuredClone(hooksJson);
  next.hooks ??= {};
  for (const event of CURSOR_EVENTS) {
    const entry = {
      command: bridgePath,
      env: { CURSOR_HOOK_EVENT: event, COMBRIEF_APP_ID: appId, COMBRIEF_MARKER: MARKER },
    };
    const list = next.hooks[event] ?? [];
    if (!list.some((h) => h.env?.COMBRIEF_MARKER === MARKER)) {
      list.push(entry);
    }
    next.hooks[event] = list;
  }
  return next;
}

export function removeCursorBridge(hooksJson: typeof injectCursorBridge extends (a: infer A) => unknown ? A : never) {
  const next = structuredClone(hooksJson);
  for (const event of Object.keys(next.hooks ?? {})) {
    next.hooks[event] = (next.hooks[event] ?? []).filter(
      (h) => h.env?.COMBRIEF_MARKER !== MARKER,
    );
  }
  return next;
}
```

import { Tray, Menu, nativeImage, Notification } from 'electron';
import type { LightStatus } from './state-machine';

export class TrayManager {
  private trays = new Map<string, Tray>();
  private frames = new Map<string, number>();

  ensureTray(appId: string, title: string): Tray {
    if (this.trays.has(appId)) return this.trays.get(appId)!;
    const tray = new Tray(nativeImage.createEmpty());
    tray.setToolTip(title);
    this.trays.set(appId, tray);
    return tray;
  }

  setStatus(appId: string, status: LightStatus, label: string): void {
    const tray = this.trays.get(appId);
    if (!tray) return;
    tray.setImage(getIcon(status, this.frames.get(appId) ?? 0));
    tray.setContextMenu(Menu.buildFromTemplate([
      { label: `${label}: ${status}`, enabled: false },
      { type: 'separator' },
      { label: '重新安装 Hooks', click: () => { /* ipc reinstall */ } },
      { label: '移除', click: () => { /* ipc uninstall */ } },
    ]));
  }

  tickAnimations(): void {
    for (const [appId, tray] of this.trays) {
      // increment frame per app based on current status
    }
  }

  notify(displayName: string): void {
    if (!Notification.isSupported()) return;
    new Notification({
      title: `${displayName} 需要你`,
      body: 'Agent 已完成，等待你的输入或确认',

```typescript
import { app, BrowserWindow } from 'electron';
import { loadConfig, saveConfig, combriefHome } from './config';
import { createCombriefServer } from './http-server';
import { TrayManager } from './tray-manager';
import { AppController } from './app-controller';

app.whenReady().then(async () => {
  const cfg = loadConfig();
  saveConfig(combriefHome(), cfg);

  const trayManager = new TrayManager();
  const controller = new AppController(cfg, trayManager);

  const server = createCombriefServer({
    port: cfg.port,
    token: cfg.token,
    registeredApps: new Set(cfg.apps),
    onState: (p) => controller.handleState(p),
  });
  await new Promise<void>((r) => server.listen(cfg.port, '127.0.0.1', () => r()));

  setInterval(() => controller.tickTimeouts(), 1000);
  setInterval(() => trayManager.tickAnimations(), 100);

  for (const appId of cfg.apps) {
    trayManager.ensureTray(appId, appId);
    controller.setTray(appId, 'offline');
  }

  // 打开设置窗口 — Task 12
});

app.on('window-all-closed', (e) => e.preventDefault()); // 托盘常驻
```
